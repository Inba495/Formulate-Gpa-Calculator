    <script>
        function switchTab(target) {
            // lazy tab switcher, just updates active state directly
            const links = document.getElementsByClassName('tab-link');
            links[0].classList.remove('active');
            links[1].classList.remove('active');
            
            document.getElementById('panel-gpa').style.display = 'none';
            document.getElementById('panel-geo').style.display = 'none';

            if(target === 'gpa') {
                links[0].classList.add('active');
                document.getElementById('panel-gpa').style.display = 'block';
            } else {
                links[1].classList.add('active');
                document.getElementById('panel-geo').style.display = 'block';
            }
        }

        function toggleShapeInputs() {
            var type = document.getElementById('shape-select').value;
            if(type === 'torus') {
                document.getElementById('wrap-torus').style.display = 'block';
                document.getElementById('wrap-sphere').style.display = 'none';
            } else {
                document.getElementById('wrap-torus').style.display = 'none';
                document.getElementById('wrap-sphere').style.display = 'block';
            }
        }

        function runGPACalc() {
            // grabbed these directly inside the runner instead of storing clean globals
            var pGpa = parseFloat(document.getElementById('prev-gpa').value) || 0;
            var pCreds = parseFloat(document.getElementById('prev-creds').value) || 0;
            var tPts = parseFloat(document.getElementById('term-pts').value) || 0;
            var tCreds = parseFloat(document.getElementById('term-creds').value) || 0;

            var box = document.getElementById('gpa-out');
            box.style.display = 'block';
            box.style.color = '#fff';

            if (pGpa < 0 || pGpa > 4 || pCreds < 0 || tPts < 0 || tCreds < 0) {
                box.style.color = '#fc8181';
                box.innerText = "Error: Negative numbers or impossible GPA entered.";
                return;
            }

            var totalCredits = pCreds + tCreds;
            if(totalCredits === 0) {
                box.innerText = "No credits calculated yet.";
                return;
            }

            var finalGpa = ((pGpa * pCreds) + tPts) / totalCredits;
            
            // output formats directly to HTML string clumsily
            box.innerHTML = "<strong>New Cum GPA:</strong> " + finalGpa.toFixed(3) + "<br><span style='font-size:12px;color:#a0aec0;'>Total credits recorded: " + totalCredits + "</span>";
        }

        function runVolumeCalc() {
            var shape = document.getElementById('shape-select').value;
            var box = document.getElementById('geo-out');
            box.style.display = 'block';
            box.style.color = '#fff';

            if(shape === 'torus') {
                var R = parseFloat(document.getElementById('torus-R').value);
                var r = parseFloat(document.getElementById('torus-r').value);

                if(isNaN(R) || isNaN(r) || R <= 0 || r <= 0) {
                    box.style.color = '#fc8181';
                    box.innerText = "Must provide positive values.";
                    return;
                }
                if(r >= R) {
                    box.style.color = '#fc8181';
                    box.innerText = "Minor radius can't be bigger than major radius.";
                    return;
                }

                // plain text pi values because why not
                var vol = 2 * Math.PI * Math.PI * R * (r * r);
                box.innerHTML = "Volume calculation: <strong>" + vol.toFixed(4) + "</strong>";
            } else {
                var rad = parseFloat(document.getElementById('sphere-radius').value);
                if(isNaN(rad) || rad <= 0) {
                    box.style.color = '#fc8181';
                    box.innerText = "Invalid radius metric.";
                    return;
                }
                var vol = (4/3) * Math.PI * Math.pow(rad, 3);
                box.innerHTML = "Volume calculation: <strong>" + vol.toFixed(4) + "</strong>";
            }
        }
